Table clientes {
  id int [pk, increment]
  nombre varchar(100) [not null]
  estado enum('ACTIVO', 'INACTIVO') [not null]
  limite_credito decimal(10,2) [not null]
}

Table direcciones {
  id int [pk, increment]
  cliente_id int [not null]
  calle varchar(150) [not null]
  ciudad varchar(100) [not null]
  estado_provincia varchar(100) [not null]
  codigo_postal varchar(20) [not null]
}

Table envios {
  id int [pk, increment]
  cliente_id int [not null]
  direccion_origen_id int [not null]
  direccion_destino_id int [not null]
  costo_total decimal(10,2) [not null]
  fecha datetime [not null]
  estado_global enum('REGISTRADO', 'EN_TRANSITO', 'ENTREGADO', 'CANCELADO') [not null]
}

Table paquetes {
  id int [pk, increment]
  envio_id int [not null]
  almacen_id int [not null]
  peso decimal(8,2) [not null]
  dimensiones varchar(50) [not null]
  contenido text [not null]
}

Table almacenes {
  id int [pk, increment]
  nombre varchar(100) [not null]
  ubicacion varchar(150) [not null]
  capacidad_maxima int [not null]
}

Table historial_rastreo {
  id int [pk, increment]
  envio_id int [not null]
  fecha datetime [not null]
  ubicacion_actual varchar(150) [not null]
  estado_evento varchar(50) [not null]
  observaciones text
}

Table auditoria_envios {
  id int [pk, increment]
  envio_id int [not null]
  costo_anterior decimal(10,2) [not null]
  costo_nuevo decimal(10,2) [not null]
  usuario varchar(100) [not null]
  fecha_modificacion datetime [not null]
}


Ref: direcciones.cliente_id > clientes.id
Ref: envios.cliente_id > clientes.id
Ref: envios.direccion_origen_id > direcciones.id
Ref: envios.direccion_destino_id > direcciones.id
Ref: paquetes.envio_id > envios.id
Ref: paquetes.almacen_id > almacenes.id
Ref: historial_rastreo.envio_id > envios.id
Ref: auditoria_envios.envio_id > envios.id
